cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "CUDA compute capabilities")
endif()
project(ComputeLab LANGUAGES CXX CUDA)
find_package(CUDAToolkit 12.8 REQUIRED)
add_executable(compute_lab main.cu)
target_compile_features(compute_lab PRIVATE cxx_std_17 cuda_std_17)
target_link_libraries(compute_lab PRIVATE CUDA::cudart CUDA::cublas)
target_compile_options(compute_lab PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
target_compile_definitions(compute_lab PRIVATE LAB_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")

include(CTest)
if(BUILD_TESTING)
  add_test(NAME compute_gemm_edges COMMAND compute_lab gemm 19 33 7)
  add_test(NAME compute_gemm_projections COMMAND compute_lab gemm 257 193 129)
  add_test(NAME compute_dot COMMAND compute_lab gemm 1 1 257)
  add_test(NAME compute_digits COMMAND compute_lab digits 60)
  add_test(NAME compute_chain COMMAND compute_lab chain 33 19 7)
  add_test(NAME compute_partial_block COMMAND compute_lab reduce 261)
  add_test(NAME compute_full_ink COMMAND compute_lab reduce 115008)
  add_test(NAME compute_graph COMMAND compute_lab spmv "${CMAKE_CURRENT_SOURCE_DIR}/data/karate.mtx" 3)
  add_test(NAME compute_roof COMMAND compute_lab spmv "${CMAKE_CURRENT_SOURCE_DIR}/data/bcsstk14.mtx" 3)
  add_test(NAME compute_poisson_odd COMMAND compute_lab poisson 33 19 5)
  add_test(NAME compute_poisson_even COMMAND compute_lab poisson 31 21 12)
endif()
